//DEFAULT PARAMETER
function greet(name = "Guest") {   --> default (means if no parameter mentioned --> default params will be taken)
    console.log(`Welcome ${name}!`);
}

greet("Jai") //Welcome Jai
greet("Udhaya") //Welcome Udhaya
greet() // Welcome Guest

//SPREAD OPERATOR
const num = [1, 2, 3, 4, 5]; 
const newNum = [...num, 6, 7, 8];  //...num adds/joins
console.log(newNum);

// DECONSTRUCTOR
 
 normal way:
const arr = ["Aryan", "Arush", "Vivek", "Rohit"];
const first = arr[0];
const second = arr[1];
const third = arr[2];
const fourth = arr[3];

deconstructor:
const [first, second, third] = ["Jai", "Udhaya", "Annie"]; // // position-based extraction
console.log(first, second, third);

const { name, age } = {name: "Jai", age: 22}; // key-based extracted
console.log(name, age);

INTERFACE & ENUM:
~~~~~~~~~~~~~~~~
    -Declaration without body
    -(not initialization)
syntax:
interface person {
    name: string;
    role: Role; //enum
    isAdmin?: boolean; // optional property ?
}
enum Role{
    Admin = "ADMIN",
    User = "USER",
    Guest = "GUEST"
}
const person1: person = {
    id: 1,
    name: "Diana",
    age: 25,
    role: Role.Admin,
}
const person2: person = {
    name: "Annie",
    role: Role.User,
}
console.log(person1, person2);

TUPLE:
~~~~~~
    -specific types [num, num] or [num, str]
Syntax:
let nums: [number, number] = [1, 2];
console.log(nums);

let user: [number, string] = [1, "Diana"];
console.log(user);

GENERICS:
~~~~~~~~~
    -function returns the same type it receives
interface Bag<T> {
    content: T;
} 
const stringBag: Bag<string> = { content: "Book" };
const numberBag: Bag<number> = { content: 6 };
console.log(stringBag, numberBag);

O/P: { content: 'Book' } { content: 6 }
