--> const abc = Object.create(ABC); // creates a new object abc with prototype ABC

EG:
const ABC = { HELLO: true };
const abc = Object.create(ABC); // creates a new object abc with prototype ABC
abc.hello = () => console.log("hello");
console.log(abc.HELLO); // inherted from ABC //TRUE
console.log(abc.hasOwnProperty('HELLO')) //FALSE
console.log(abc.hasOwnProperty('hello')) //TRUE
console.log(abc.toString()) //o/p: [object Object]
