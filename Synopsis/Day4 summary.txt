function getData() {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            resolve("Data received"); //resolve->then , reject->catch
        }, 2000);
    });
}
 
getData().then((data) => {
    console.log(data);
}).catch((error) => {
    console.error("Error:", error);
});
